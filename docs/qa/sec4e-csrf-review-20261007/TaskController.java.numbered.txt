  12 @RestController

  56 
  57     @PostMapping("/{id}/process")
  58     public ResponseEntity<String> processTask(@PathVariable Long id,
  59                                                @RequestParam String action,
  60                                                @RequestParam(defaultValue = "USER") String role,
  61                                                @RequestParam(defaultValue = "false") boolean urgent,
  62                                                @RequestParam(defaultValue = "false") boolean bulk) {

  64                 .orElseThrow(() -> new RuntimeException("Task not found"));
